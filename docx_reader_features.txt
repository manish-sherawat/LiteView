<script>
const steps = [
  {label:'Loading library…',   pct:'18%'},
  {label:'Loading fonts…',     pct:'36%'},
  {label:'Syncing bookmarks…', pct:'55%'},
  {label:'Restoring session…', pct:'78%'},
  {label:'Ready',              pct:'100%'},
];
const items = document.querySelectorAll('.step-item');
const lbl   = document.getElementById('prog-label');
const pct   = document.getElementById('prog-pct');
let cur = 0;

const advance = () => {
  if (cur > 0) {
    items[cur-1].classList.remove('active');
    items[cur-1].classList.add('done');
  }
  if (cur < items.length) {
    items[cur].classList.add('active');
    lbl.textContent = steps[cur].label;
    pct.textContent = steps[cur].pct;
    cur++;
    if (cur < steps.length) setTimeout(advance, 700);
  }
};
setTimeout(advance, 1100);
</script>
